// --- Progression, Vines & Achievements Engine --- const LAYER_IDS = { 1: ['vine_climber', 'phonics_swamp', 'canopy_crosswords', 'root_word_expedition', 'anagram_antics', 'syllable_chop', 'homophone_haven', 'grammar_grove', 'spelling_bee_swarm', 'context_clue_quest'], 2: ['canopy_count', 'termite_trail', 'fraction_foliage', 'perimeter_patrol', 'times_table_treetops', 'negative_nesting', 'ratio_river', 'angle_archer', 'number_line_leap', 'jungle_budget'], 3: ['canopy_circuit', 'food_web_weaver', 'deduction_trail', 'river_crossing', 'pattern_pioneer', 'pathfinding_drone', 'resource_balancer', 'optical_canopy', 'sorting_matrix', 'the_false_log'] }; function saveStudentName(name) { localStorage.setItem('jacksjungle_student_name', name.trim()); } function updateProgress() { // 1. Scan unique completed games across all tiers let completedCount = 0; let layer1Done = 0, layer2Done = 0, layer3Done = 0; const allNodes = document.querySelectorAll('.game-node[data-game-id]'); allNodes.forEach(node => { const id = node.getAttribute('data-game-id'); const isDone = localStorage.getItem(`jacksjungle_completed_${id}`) === 'true'; if (isDone) { completedCount++; if (LAYER_IDS[1].includes(id)) layer1Done++; if (LAYER_IDS[2].includes(id)) layer2Done++; if (LAYER_IDS[3].includes(id)) layer3Done++; // Inject completion root overlay if not already present if (!node.querySelector('.root-overlay')) { node.insertAdjacentHTML('beforeend', rootSvgTemplate); const foot = node.querySelector('.game-footer-row'); if (foot && !foot.querySelector('.completed-badge')) { const b = document.createElement('span'); b.className = 'completed-badge'; b.innerHTML = '🌿 Completed'; foot.appendChild(b); } } } }); // 2. Derive true leaves from completed games (20 leaves per unique game) // Or fall back to manual leaf counter if higher let manualLeaves = parseInt(localStorage.getItem('jacksjungle_leaves'), 10) || 0; let calculatedLeaves = completedCount * 20; let leaves = Math.max(manualLeaves, calculatedLeaves); localStorage.setItem('jacksjungle_leaves', leaves.toString()); // 3. Update Leaf Readout and Canopy Density Bar const readout = document.getElementById('leaf-readout'); if (readout) readout.textContent = `🍃 ${leaves} Leaves`; // Scale density: 10 games = 50%, 20 games = 100% const percent = Math.min(100, Math.round((completedCount / 20) * 100)); const progressBar = document.getElementById('leaf-progress-bar'); if (progressBar) progressBar.style.width = `${percent}%`; const percentLabel = document.getElementById('progress-percent-label'); if (percentLabel) percentLabel.textContent = `Canopy Density: ${percent}%`; // 4. Hide Hero CTA if any game has been completed const hero = document.getElementById('hero-cta'); if (hero) { if (completedCount > 0) hero.classList.add('hidden'); else hero.classList.remove('hidden'); } // 5. Creep Vines on sides const vineGrowth = Math.min(1, completedCount / 12); const leftVine = document.getElementById('left-vine'); const rightVine = document.getElementById('right-vine'); if (leftVine && rightVine) { leftVine.style.opacity = vineGrowth; leftVine.style.transform = `translateX(${-40 + (vineGrowth * 40)}px)`; rightVine.style.opacity = vineGrowth; rightVine.style.transform = `translateX(${40 - (vineGrowth * 40)}px) scaleX(-1)`; } // 6. Scale falling leaf particle density const targetParticleCount = Math.floor(baseParticleCount + (completedCount * 3)); if (particles.length !== targetParticleCount) { initParticles(targetParticleCount); } // 7. Layer 2 Unlock Check (50 Leaves or 3 Layer 1 games) if (leaves >= 50 || layer1Done >= 3) { const req2 = document.getElementById('understory-req'); if (req2) { req2.textContent = "Level 2 • Unlocked"; req2.style.color = "var(--accent)"; } document.querySelectorAll('#understory-grid .game-node').forEach(node => { node.classList.remove('locked'); const banner = node.querySelector('.lock-banner'); if (banner) { banner.textContent = "Understory • Unlocked"; banner.style.color = "var(--accent)"; } }); } // 8. Layer 3 Unlock Check (120 Leaves or 6 total games) if (leaves >= 120 || completedCount >= 6) { const req3 = document.getElementById('emergent-req'); if (req3) { req3.textContent = "Level 3 • Unlocked"; req3.style.color = "var(--accent)"; } document.querySelectorAll('#emergent-grid .game-node').forEach(node => { node.classList.remove('locked'); const banner = node.querySelector('.lock-banner'); if (banner) { banner.textContent = "Canopy • Unlocked"; banner.style.color = "var(--accent)"; } }); } // 9. Update Certificate Button States const c1 = document.getElementById('cert-btn-1'); const c2 = document.getElementById('cert-btn-2'); const c3 = document.getElementById('cert-btn-3'); const cM = document.getElementById('cert-btn-master'); if (c1) c1.disabled = (layer1Done < 10); if (c2) c2.disabled = (layer2Done < 10); if (c3) c3.disabled = (layer3Done < 10); if (cM) cM.disabled = (completedCount < 30); } // Ensure execution on initial load, pageshow (bfcache), and storage events window.addEventListener('DOMContentLoaded', updateProgress); window.addEventListener('pageshow', updateProgress); window.addEventListener('focus', updateProgress); window.addEventListener('storage', updateProgress);
